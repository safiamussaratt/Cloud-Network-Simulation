    def create_topology(self):
        self.G = nx.Graph()
        net = self.config["network"]
        links = self.config["links"]

        routers = [f"Router{i+1}" for i in range(net["num_routers"])]
        datacenters = [f"DC{i+1}" for i in range(net["num_datacenters"])]
        users = [f"User{i+1}" for i in range(net["num_users"])]

        for r in routers:
            self.G.add_node(r, type="router", capacity=10000)
        for dc in datacenters:
            self.G.add_node(dc, type="datacenter", capacity=5000)
        for u in users:
            self.G.add_node(u, type="user", capacity=100)

        if "custom_connections" in links:
            for u, v in links["custom_connections"]:
                self.G.add_edge(
                    u, v,
                    bandwidth=links["user_bandwidth"],
                    delay=links["user_delay"],
                    utilization=0,
                    traffic_volume=0
                )
            return self.G

        for i in range(len(routers) - 1):
            self.G.add_edge(
                routers[i], routers[i+1],
                bandwidth=links["backbone_bandwidth"],
                delay=links["backbone_delay"],
                utilization=0,
                traffic_volume=0
            )

        for dc in datacenters:
            self.G.add_edge(
                routers[0], dc,
                bandwidth=links["dc_bandwidth"],
                delay=links["dc_delay"],
                utilization=0,
                traffic_volume=0
            )

        for i, user in enumerate(users):
            ridx = i % len(routers)
            self.G.add_edge(
                routers[ridx], user,
                bandwidth=links["user_bandwidth"],
                delay=links["user_delay"],
                utilization=0,
                traffic_volume=0
            )

        return self.G

    def calculate_node_positions(self):
        pos = {}
        routers = [n for n in self.G.nodes() if self.G.nodes[n]["type"] == "router"]
        dcs = [n for n in self.G.nodes() if self.G.nodes[n]["type"] == "datacenter"]
        users = [n for n in self.G.nodes() if self.G.nodes[n]["type"] == "user"]

        for i, r in enumerate(routers):

        for i, u in enumerate(users):

        return pos
    def generate_flows(self):
        traffic_config = self.config["traffic"]
        flows = []
        
        datacenters = [n for n in self.G.nodes() if self.G.nodes[n]['type'] == 'datacenter']
        users = [n for n in self.G.nodes() if self.G.nodes[n]['type'] == 'user']
        
        for user in users:
            for dc in datacenters[:2]:
                flow = {
                    'id': f"{user}-{dc}",
                    'source': user,
                    'destination': dc,
                    'packet_size': traffic_config["packet_size"],
                    'rate': random.choice(traffic_config["flow_rates"]),
                    'active': True,
                    'start_time': 0,
                    'lifetime': float('inf')
                }
                flows.append(flow)
        
        for i in range(len(datacenters) - 1):
            flow = {
                'id': f"{datacenters[i]}-{datacenters[i+1]}",
                'source': datacenters[i],
                'destination': datacenters[i+1],
                'packet_size': traffic_config["packet_size"],
                'rate': random.choice([800, 1000, 1200]),
                'active': True,
                'start_time': 0,
                'lifetime': float('inf')
            }
            flows.append(flow)
            
        return flows
    
    def calculate_path(self, source, destination):
        try:
            def weight_function(u, v, edge_attrs):
                base_delay = edge_attrs.get('delay', 10)
                utilization = edge_attrs.get('utilization', 0.0)
                utilization_penalty = utilization * 100
                return base_delay + utilization_penalty
            
            return nx.shortest_path(self.G, source, destination, weight=weight_function)
        except nx.NetworkXNoPath:
            try:
                return nx.shortest_path(self.G, source, destination)
            except:
                return []
        except:
            return []
    
    def simulate_packet_flow(self, flows, current_time):
        packets = []
        traffic_config = self.config["traffic"]
        packet_prob = traffic_config.get("packet_generation_probability", 0.3)
        
        for flow in flows:
            if not flow['active']:
                continue
                
            packets_per_interval = (flow['rate'] / flow['packet_size']) / 10
            if random.random() < min(packets_per_interval * packet_prob, 0.8):
                path = self.calculate_path(flow['source'], flow['destination'])
                if len(path) >= 2:
                    packet_size = flow['packet_size'] * random.uniform(0.8, 1.2)
                    packet = {
                        'flow_id': flow['id'],
                        'path': path,
                        'size': packet_size,
                        'start_time': current_time,
                        'current_hop': 0,
                        'delivered': False
                    }
                    packets.append(packet)
        
        return packets
    
    def update_link_utilization(self, packets):
        for u, v in self.G.edges():
            self.G[u][v]['traffic_volume'] = 0
        
        link_traffic = defaultdict(int)
        for packet in packets:
            path = packet['path']
            for i in range(len(path) - 1):
                u, v = sorted([path[i], path[i+1]])
                link_traffic[(u, v)] += packet['size']
        
        for (u, v), traffic in link_traffic.items():
            self.G[u][v]['traffic_volume'] = traffic
            bandwidth = self.G[u][v]['bandwidth']
            new_utilization = traffic / bandwidth
            
            old_utilization = self.G[u][v]['utilization']
            smoothed_utilization = old_utilization * 0.7 + new_utilization * 0.3
            self.G[u][v]['utilization'] = min(smoothed_utilization, 1.0)

    def run_simulation(self):
        print("Starting Cloud Network Simulation...")
        
        print("Creating network topology...")
        self.G = self.create_topology()
        self.pos = self.calculate_node_positions()
        
        print("Generating initial traffic flows...")
        flows = self.generate_flows()
        print(f"   Created {len(flows)} initial flows")
        
        flow_metrics = defaultdict(lambda: {
            'throughput': deque(maxlen=100),
            'latency': deque(maxlen=100),
            'packet_loss': deque(maxlen=100),
            'congestion_window': deque(maxlen=100)
        })
        
        self.link_utilization_history = defaultdict(list)
        metrics_history = []
        duration = self.config["network"]["simulation_duration"]
        
        print("Running simulation...")
        for current_time in range(duration):
            self.flow_manager.manage_flows_dynamically(flows, current_time)
            
            packets = self.simulate_packet_flow(flows, current_time)
            
            for u, v in self.G.edges():
                self.link_utilization_history[(u, v)].append(self.G[u][v]['utilization'])
            
            self.update_link_utilization(packets)
            
            for flow in flows:
                if not flow['active']:
                    continue
                    
                flow_id = flow['id']
                
                base_throughput = flow['rate']
                throughput = max(0, base_throughput + random.uniform(-50, 50))
                
                path = self.calculate_path(flow['source'], flow['destination'])
                base_latency = len(path) * 5
                congestion_latency = sum(self.G[u][v]['utilization'] * 20 for u, v in zip(path[:-1], path[1:])) if len(path) > 1 else 0
                latency = base_latency + congestion_latency + random.uniform(-2, 2)
                
                max_utilization = max((self.G[u][v]['utilization'] for u, v in zip(path[:-1], path[1:])), default=0)
                packet_loss = max_utilization * 5 if max_utilization > 0.7 else 0
                
                if len(flow_metrics[flow_id]['congestion_window']) == 0:
                    cwnd = 10
                else:
                    cwnd = flow_metrics[flow_id]['congestion_window'][-1]
                    if random.random() < max_utilization * 0.1:
                        cwnd = max(1, cwnd // 2)
                    else:
                        cwnd = min(64, cwnd + 1)
                
                flow_metrics[flow_id]['throughput'].append(throughput)
                flow_metrics[flow_id]['latency'].append(latency)
                flow_metrics[flow_id]['packet_loss'].append(packet_loss)
                flow_metrics[flow_id]['congestion_window'].append(cwnd)
                
                metrics_history.append({
                    'time': current_time,
                    'flow_id': flow_id,
                    'throughput': throughput,
                    'latency': latency,
                    'packet_loss': packet_loss,
                    'congestion_window': cwnd,
                    'source': flow['source'],
                    'destination': flow['destination'],
                    'packets_sent': len(packets)
                })
            
            if current_time % 10 == 0:
                active_flows = sum(1 for flow in flows if flow['active'])
                total_packets = len(packets)
                avg_utilization = np.mean([self.G[u][v]['utilization'] for u, v in self.G.edges()])
                print(f"   Time: {current_time}/{duration}, Active flows: {active_flows}, Packets: {total_packets}, Avg Util: {avg_utilization:.3f}")
        
        print("Simulation completed!")
        
        metrics_df = pd.DataFrame(metrics_history)
        
        csv_file = "results/csv_data/network_metrics.csv"
        metrics_df.to_csv(csv_file, index=False)
        print(f"Metrics saved to: {csv_file}")
        
        self.create_visualizations(metrics_df)
        self.create_individual_plots(metrics_df)
        
        return metrics_df

    def plot_topology(self):
        plt.cla()
        
        node_colors = []
        node_sizes = []
        for node in self.G.nodes():
            node_type = self.G.nodes[node]['type']
            if node_type == 'router':
                node_colors.append('red')
                node_sizes.append(400)
            elif node_type == 'datacenter':
                node_colors.append('blue')
                node_sizes.append(500)
            else:
                node_colors.append('green')
                node_sizes.append(300)
        
        edge_colors = []
        edge_widths = []
        edge_labels = {}
        
        for u, v in self.G.edges():
            utilization = self.G[u][v]['utilization']
            edge_colors.append(plt.cm.RdYlGn_r(utilization))
            edge_widths.append(1 + utilization * 6)
            if utilization > 0.1:
                edge_labels[(u, v)] = f'{utilization:.2f}'
        
        nx.draw(self.G, self.pos, with_labels=True, node_color=node_colors,
                node_size=node_sizes, font_size=7, font_weight='bold',
                edge_color=edge_colors, width=edge_widths,
                edge_cmap=plt.cm.RdYlGn_r, edge_vmin=0, edge_vmax=1)
        
        nx.draw_networkx_edge_labels(self.G, self.pos, edge_labels=edge_labels, 
                                   font_size=6, font_weight='bold')
        
        plt.title('Network Topology & Link Utilization', fontsize=10)

class DynamicFlowManager:
    def __init__(self, simulator):
        self.simulator = simulator
        self.flow_counter = 0
        
    def manage_flows_dynamically(self, flows, current_time):
        self.create_dynamic_flows(flows, current_time)
        self.cleanup_expired_flows(flows, current_time)
    
    def create_dynamic_flows(self, flows, current_time):
        traffic_config = self.simulator.config["traffic"]
        dynamic_flow_prob = traffic_config.get("dynamic_flow_probability", 0.08)
        
        if random.random() < dynamic_flow_prob:
            self._create_random_flow(flows, current_time)
    
    def _create_random_flow(self, flows, current_time):
        users = [n for n in self.simulator.G.nodes() if self.simulator.G.nodes[n]['type'] == 'user']
        datacenters = [n for n in self.simulator.G.nodes() if self.simulator.G.nodes[n]['type'] == 'datacenter']
        routers = [n for n in self.simulator.G.nodes() if self.simulator.G.nodes[n]['type'] == 'router']
        
        if not users or not datacenters:
            return
        
        flow_type = random.choice(['user-dc', 'dc-dc', 'user-router', 'user-dc', 'user-dc'])
        
        if flow_type == 'user-dc':
            source = random.choice(users)
            destination = random.choice(datacenters)
        elif flow_type == 'dc-dc' and len(datacenters) >= 2:
            source, destination = random.sample(datacenters, 2)
        else:
            source = random.choice(users)
            destination = random.choice(routers)
        
        path = self.simulator.calculate_path(source, destination)
        if len(path) < 2:
            return
        
        self.flow_counter += 1
        flow_id = f"dynamic_flow_{self.flow_counter}"
        
        traffic_config = self.simulator.config["traffic"]
        lifetime_min = traffic_config.get("flow_lifetime_min", 10)
        lifetime_max = traffic_config.get("flow_lifetime_max", 30)
        
        new_flow = {
            'id': flow_id,
            'source': source,
            'destination': destination,
            'packet_size': traffic_config["packet_size"],
            'rate': random.choice(traffic_config["flow_rates"]),
            'active': True,
            'start_time': current_time,
            'lifetime': random.randint(lifetime_min, lifetime_max),
            'dynamic': True
        }
        
        flows.append(new_flow)
        print(f"Dynamic flow: {source} → {destination} (rate: {new_flow['rate']} Mbps, life: {new_flow['lifetime']}s)")
    
    def cleanup_expired_flows(self, flows, current_time):
        for flow in flows[:]:
            if flow.get('dynamic', False) and flow['active']:
                if current_time - flow['start_time'] >= flow['lifetime']:
                    flow['active'] = False
                    print(f"Flow expired: {flow['id']}")
